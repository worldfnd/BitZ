\section{Relations, reductions, and proof objects}
\label{sec:encoding}

\begingroup\footnotesize

An encoded relation contains triples $(x,O;\mathsf{wit})$.  The explicit
instance is $x$, the encoded oracle is $O$, and $\mathsf{wit}$ is the witness
view used by that relation.  Every relation below carries the same public
instance parameters, matrix $\bm M$, oracle $O$, and root $\rho=\Root(O)$.
The binding invariant is always
\begin{equation}
 O=\Enc_{\mathcal C}(\Pack(F)),\qquad
 \operatorname{vec}(F)=\pi_2(\bm f).
 \label{eq:encoded-oracle-binding}
\end{equation}
No intermediate relation replaces $F$ by $H$ in
Equation~\eqref{eq:encoded-oracle-binding}.

The edge notation distinguishes deterministic claim changes from interactive
reductions:
\begin{equation}
 \mathcal R_0\equiv_G\mathcal R_1
 \quad\text{and}\quad
 \mathcal R_0\xRightarrow{\Pi}\mathcal R_1.
 \label{eq:reduction-edge-notation}
\end{equation}
The first edge applies a public deterministic map $G$ and sends no message.
The second edge is implemented by the named protocol.

\subsection{Relation chain}

F2Z begins from the application-supplied separable Lin-BitsDual claim:
\begin{align}
 \mathcal R_{\rm LinBitsDual}[\bm h]:=
 \Bigl\{((\bm a,\bm a',\mu,\bm M),O;\bm f):\;&
  \operatorname{vec}(H)=\bm M\pi_2(\bm f),\notag\\[-2pt]
 &\mu=\sum_c a'_c\sum_r a_rH[r,c],\notag\\[-2pt]
 &O=\Enc_{\mathcal C}(\Pack(F)),
  \quad\operatorname{vec}(F)=\pi_2(\bm f)\Bigr\}.
 \label{eq:rel-linbits-h}
\end{align}
The F2Z statement binds $(\bm a,\bm a',\mu)$, the active layout $H$, the
chunk schedule, and the virtualization matrix to the transcript.  The
application proof establishing this claim is outside F2Z, and no second
commitment is created.

For each active chunk $\ell$, define
\begin{equation}
 Q_c^{(\ell)}(H)
 :=\prod_r\left(1+
       (g^{\gamma_r^{(\ell)}}-1)H[r,c]\right)
   -g^{\mu_c^{(\ell)}}.
 \label{eq:chunk-pesat-polynomial}
\end{equation}
The product relation is
\begin{align}
 \mathcal R_{\rm PESAT}[\bm h]:=
 \Bigl\{((\bm\gamma^{(\ell)},\bm\mu^{(\ell)},\bm M),O;\bm f):\;&
  \operatorname{vec}(H)=\bm M\pi_2(\bm f),
  \quad Q_c^{(\ell)}(H)=0\ (\forall c,\ell),\notag\\[-2pt]
 &O=\Enc_{\mathcal C}(\Pack(F)),
  \quad\operatorname{vec}(F)=\pi_2(\bm f)\Bigr\}.
 \label{eq:rel-pesat-h}
\end{align}
The exact folds $\mu_c^{(\ell)}$ are checked against the supplied linear claim by
Equation~\eqref{eq:fold-reconstruction} before this relation is accepted.

GKR reduces the product computation to a weighted sum claim.  Its verifier
derives public weights $\bm\omega_{\xi}$ and a target $\sigma$:
\begin{equation}
 \mathcal R_{\rm sum}^{h}:=
 \left\{((\bm\omega_{\xi},\sigma,\bm M),O;\bm f):
 \sum_z\omega_{\xi,z}H[z]=\sigma,
 \quad\operatorname{vec}(H)=\bm M\pi_2(\bm f),
 \quad\text{Equation~\eqref{eq:encoded-oracle-binding}}
 \right\}.
 \label{eq:rel-sum-h}
\end{equation}
GKR reads the active witness through its circuit oracle.  It does not query
$O$ or assert that $O$ encodes $H$.

The row protocol $\mathsf{LinCheck}_H$ reduces the weighted sum claim to one
MLE evaluation:
\begin{equation}
 \mathcal R_{\rm eval}^{h}:=
 \left\{((\bm r_h,\eta,\bm M),O;\bm f):
  \widetilde H(\bm r_h)=\eta,
  \quad\operatorname{vec}(H)=\bm M\pi_2(\bm f),
  \quad\text{Equation~\eqref{eq:encoded-oracle-binding}}
 \right\}.
 \label{eq:rel-eval-h}
\end{equation}
For $z\in\bitsset^{m_h}$ and $\bm r_h\in\K^{m_h}$, define
\begin{equation}
 \eq(z,\bm r_h):=
 \prod_{j=0}^{m_h-1}
 \bigl((1-z_j)(1-r_{h,j})+z_jr_{h,j}\bigr).
 \label{eq:equality-polynomial}
\end{equation}
By the definition of the multilinear extension,
\begin{equation}
 \widetilde H(\bm r_h)
 =\sum_{z\in\bitsset^{m_h}}\eq(z,\bm r_h)H[z].
 \label{eq:active-mle-expansion}
\end{equation}
Thus the MLE claim has coefficient vector
$\bm a_h[z]:=\eq(z,\bm r_h)$.  Scalar extension and the adjoint give
\begin{equation}
 \bm a_f:=\bm M_{\K}^{\mathsf T}\bm a_h,
 \qquad
 \langle\bm a_h,\bm h_{\K}\rangle_{\K}
 =\langle\bm a_f,\bm f_{\K}\rangle_{\K}=\eta.
 \label{eq:virtual-adjoint-rewrite}
\end{equation}
This step sends no message.  The public instance must provide a deterministic
full-adjoint evaluator, for example through
\begin{equation}
 \widetilde{\bm M_{\K}^{\mathsf T}\bm a_h}(\bm r_f)
 =\left\langle
    \bm a_h,
    \bm M_{\K}\bigl(\eq(\cdot,\bm r_f)\bigr)
  \right\rangle_{\K}.
 \label{eq:virtual-adjoint-evaluator}
\end{equation}
The resulting encoded relation is
\begin{align}
 \mathcal R_{\rm Lin}^{f}:=
 \Bigl\{((\bm a_f,\eta),O;\bm f):\;&
  \bm a_f\in\K^{n_f},
  \quad\langle\bm a_f,\bm f_{\K}\rangle_{\K}=\eta,\notag\\[-2pt]
 &O=\Enc_{\mathcal C}(\Pack(F)),
  \quad\operatorname{vec}(F)=\pi_2(\bm f)\Bigr\}.
 \label{eq:rel-lin-f}
\end{align}

\subsection{Reductions}

The complete relation graph is
\begin{equation}
\begin{aligned}
 \mathcal R_{\rm LinBitsDual}[\bm h]
 &\xRightarrow{\Pi_1}
 \mathcal R_{\rm PESAT}[\bm h]
 \xRightarrow{\Pi_{\rm GKR}}
 \mathcal R_{\rm sum}^{h}
 \\[2pt]
 &\xRightarrow{\Pi_{\rm MLE}}
 \mathcal R_{\rm eval}^{h}
 \equiv_{\bm M^{\mathsf T}}
 \mathcal R_{\rm Lin}^{f}
 \xRightarrow{\Pi_{\rm iopp}}
 \mathcal R_{\rm triv}.
\end{aligned}
\label{eq:solid-reduction-graph}
\end{equation}

\paragraph{Integer folds and products.}
$\Pi_1$ sends the active exact folds
\begin{equation}
 \pi_1:=
 \bigl(\bm\nu^{(\ell)},\bm\mu^{(\ell)}\bigr)_{
       \ell\in\mathcal A\uparrow},
 \qquad
 \nu_c^{(\ell)}:=g^{\mu_c^{(\ell)}}.
 \label{eq:pi1-chunk-proof}
\end{equation}
The verifier requires $0\le\mu_c^{(\ell)}<2^{127}$, checks
$g^{\mu_c^{(\ell)}}=\nu_c^{(\ell)}$, and checks
Equation~\eqref{eq:fold-reconstruction} before deriving the product
instances.  The vectors $\bm\nu^{(\ell)}$ are proof records.  The integer
vectors $\bm\mu^{(\ell)}$ are checked, non-absorbed hints.  Inactive chunks
have derived value zero.

\paragraph{GKR and row LinCheck.}
For every active chunk, the fixed GKR interface consumes one product instance
and returns one weighted sum claim on $H$:
\begin{equation}
 (\tr',\bm\omega_{\xi},\sigma)
 \leftarrow
 \mathsf{GKR.Verify}
 (\mathsf{pp}_{\rm GKR},\mathcal I_{\rm GKR},
  \pi_{\rm GKR};\tr).
 \label{eq:gkr-verifier-contract}
\end{equation}
The row LinCheck then consumes that sum claim and returns
$(\bm r_h,\eta)$ for Equation~\eqref{eq:rel-eval-h}.  Write its proof fragment
as $\pi_{\rm MLE}$.  The existing GKR wire record carries the ordered pair
$(\pi_{\rm GKR},\pi_{\rm MLE})$; this semantic split does not add a wire tag or
change framing.

\paragraph{Adjoint and opening.}
The verifier evaluates $\bm a_f=\bm M_{\K}^{\mathsf T}\bm a_h$ locally.
If $\bm a_f$ is already MLE-shaped, $\Pi_{\rm iopp}$ proceeds directly to the
ring switch.  Otherwise it first runs the coefficient protocol
$\mathsf{LinCheck}_f$.  The decision is determined by the public instance and
coefficient shape, not by a prover flag.

\subsection{Opening a pulled-back claim}

For a general coefficient, the coefficient protocol $\mathsf{LinCheck}_f$
proves
\begin{equation}
 \sum_{x\in\bitsset^{m_f}}
   \widetilde a_f(x)\widetilde f_{\K}(x)=\eta.
 \label{eq:lincheck-sum}
\end{equation}
Starting with $\sigma_{-1}:=\eta$, round $i$ sends a degree-two polynomial
$g_i$ and checks
\begin{equation}
 g_i(0)+g_i(1)=\sigma_{i-1},\qquad
 r_i\leftarrow\K,\qquad
 \sigma_i:=g_i(r_i),\qquad 0\le i<m_f.
 \label{eq:lincheck-round}
\end{equation}
Let $\bm r=(\bm r_{\lo},\bm r_{\hi})$, with
$|\bm r_{\lo}|=7$.  The first ring-switch message derives
\begin{equation}
 z:=\sum_{v=0}^{127}\eq(v,\bm r_{\lo})s_v
   =\widetilde f_{\K}(\bm r),\qquad
 \sigma_{m_f-1}=\widetilde a_f(\bm r)z.
 \label{eq:lincheck-terminal}
\end{equation}
The verifier performs this check after absorbing the message and before its
next squeeze.  It uses no division and has no exceptional branch when
$\widetilde a_f(\bm r)=0$.  The coefficient-LinCheck fragment is
$\pi_{\rm LC}:=(g_i)_{i=0}^{m_f-1}$ and has error at most
$\varepsilon_{\rm LC}:=2m_f/|\K|$.  An MLE-shaped coefficient sets
$\pi_{\rm LC}=()$ and $\varepsilon_{\rm LC}=0$.

For $0\le v<128$, let $F_v$ be the $v$-th bit plane of the packed committed
layout.  The prover transmits
\begin{equation}
 \bm s=(s_v)_{v<128},\qquad
 s_v:=\widetilde F_v(\bm r_{\hi}).
 \label{eq:ring-switch-message}
\end{equation}
The verifier derives a packed claim
\begin{equation}
 O=\Enc_{\mathcal C}(P),\qquad
 \langle B,P\rangle_{\K}=\beta.
 \label{eq:packed-claim}
\end{equation}
For several active chunks, fresh coefficients $\bm\eta_{\rm bat}$ combine the
packed claims without adding a proof field:
\begin{equation}
 B:=\sum_{\ell\in\mathcal A}\eta_{{\rm bat},\ell}B_\ell,
 \qquad
 \beta:=\sum_{\ell\in\mathcal A}\eta_{{\rm bat},\ell}\beta_\ell.
 \label{eq:batch-packed-claim}
\end{equation}
The recursive constrained-code proof $\pi_{\rm lig}$ verifies this claim
against $\rho$.

\subsection{Proof object and soundness}

The semantic proof object is
\begin{equation}
 \pi_{\mathrm{F2Z}}
 =\left(
   \bigl(\bm\nu^{(\ell)},\bm\mu^{(\ell)},
         \pi_{\rm GKR}^{(\ell)},\pi_{\rm MLE}^{(\ell)}\bigr)_
         {\ell\in\mathcal A\uparrow},
   \pi_{\rm LC},
   (\bm s^{(\ell)})_{\ell\in\mathcal A\uparrow},
   \pi_{\rm lig}
 \right).
 \label{eq:proof-object}
\end{equation}

\begin{center}
\begin{tabularx}{\linewidth}{@{}l l X@{}}
\toprule
Fragment & Successor claim & Error \\
\midrule
$\pi_1$ & product claims on $H$ & $\varepsilon_1$ \\
$\pi_{\rm GKR}^{(\ell)}$ & weighted sum claim on $H$ & $\varepsilon_{\rm GKR}^{(\ell)}$ \\
$\pi_{\rm MLE}^{(\ell)}$ & MLE claim on $H$ & $\varepsilon_{\rm MLE}^{(\ell)}$ \\
$\pi_{\rm LC}$ & MLE-shaped claim on $F$ & $\varepsilon_{\rm LC}$ or $0$ \\
ring switch and batch & packed claim & $\varepsilon_{\rm ring}^{(\ell)}$, $\varepsilon_{\rm batch}$ \\
$\pi_{\rm lig}$ & acceptance & $\varepsilon_{\rm open}$ \\
\bottomrule
\end{tabularx}
\end{center}

The v1 wire format partitions transmitted fields, in replay order, as
\begin{equation}
 \pi_{\rm wire}:=(N_{\rm F2Z},H_{\rm F2Z}),\qquad
 \begin{aligned}
  \NargLen(\pi)&:=|N_{\rm F2Z}|,&
  \HintLen(\pi)&:=|H_{\rm F2Z}|,\\
  \PayloadLen(\pi)&:=|N_{\rm F2Z}|+|H_{\rm F2Z}|,&
  \WireLen(\pi)&:=|\HostEncode(N_{\rm F2Z},H_{\rm F2Z})|.
 \end{aligned}
 \label{eq:proof-channel-lengths}
\end{equation}
The byte tags and container rules are fixed in Section~\ref{sec:wire-container}.

The composed error is bounded by
\begin{equation}
 \begin{aligned}
 \varepsilon_{\rm F2Z}
 \le{}&\varepsilon_1
  +\sum_{\ell\in\mathcal A}
    \left(
      \varepsilon_{\rm GKR}^{(\ell)}
      +\varepsilon_{\rm MLE}^{(\ell)}
      +\varepsilon_{\rm ring}^{(\ell)}
    \right)\\
 &+\varepsilon_{\rm LC}
  +\varepsilon_{\rm batch}+\varepsilon_{\rm open}.
 \end{aligned}
 \label{eq:composed-soundness}
\end{equation}
