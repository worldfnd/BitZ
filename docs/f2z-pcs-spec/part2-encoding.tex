\section{Relations and direct reductions}
\label{sec:encoding}

\begingroup\footnotesize

\subsection{Relations and reduction notation}

This section defines the following five relation types, in protocol order:
\[
 \mathcal R_{\rm LinBitsDual},\quad
 \mathcal R_{\rm PESAT},\quad
 \mathcal R_{\rm sum}^{h},\quad
 \mathcal R_{\rm eval}^{h},\quad
 \LIN(\K,\mathcal C,\Pack).
\]
These relations share the committed witness, encoded oracle, developer
configuration, and public root fixed in
Sections~\ref{sec:overview}--\ref{sec:instance}.

For reductions, the notation
\begin{equation}
 \mathcal R_0\equiv_G\mathcal R_1
 \quad\text{and}\quad
 \mathcal R_0\xRightarrow{\Pi}\mathcal R_1
 \label{eq:reduction-edge-notation}
\end{equation}
distinguishes a public, zero-message rewrite $G$ from an interactive
reduction $\Pi$.  Each subsection defines one relation and then gives the
reduction to its immediate successor.  The chain stops at the encoded $\LIN$
relation; Section~\ref{sec:interaction} opens the resulting claim.

\subsection{The Lin-BitsDual relation}

F2Z receives an arbitrary linear claim over $\Fq$.  Use the canonical
$\Fq$-view $H_q=\pi_q(H_{\mathbb Z})$ from Section~\ref{sec:instance}.  The
input relation is
\begin{align}
 \mathcal R_{\rm LinBitsDual}[\bm h]:=
 \Bigl\{((\bm v,\mu,\bm M),O;\bm f):\;&
  \bm f\in\bitsset^{n_f},\quad
  \operatorname{vec}(H)=\bm M\pi_2(\bm f),\notag\\[-2pt]
 &\mu=\sum_{r\in\bitsset^{d_r}}\sum_{c\in\bitsset^{d_c}}
       v[(r,c)]H_q[r,c],\notag\\[-2pt]
 &\text{Equations~\eqref{eq:virtual-witness} and
   \eqref{eq:overview-commitment} hold}
 \Bigr\}.
 \label{eq:rel-linbits-h}
\end{align}
The PIOP that produced this claim is separate from F2Z.  From this
point onward, the protocol uses only the public Lin-BitsDual instance, the
developer configuration, and the commitment to $\bm f$.

\paragraph{Direct reduction to F2Z PESAT ($\Pi_1$).}
The successor of Lin-BitsDual is PESAT.  To retain that boundary while accepting an arbitrary public
coefficient, this specification writes
$\Pi_1:=\Pi_{1,{\rm in}};\Pi_{1,{\rm core}}$.  The first phase is the
input Sumcheck adapter, and the second is the bounded
integer-lifting and grand-product phase corresponding to the paper's core
reduction.

\smallskip
\noindent\textbf{Input Sumcheck phase.}
Let $\widetilde{\bm v}$ and $\widetilde H_q$ be the multilinear extensions of
the two tables in Equation~\eqref{eq:rel-linbits-h}.  The incoming scalar
claim is the hypercube sum
\begin{equation}
 \mu=\sum_{r\in\bitsset^{d_r}}\sum_{c\in\bitsset^{d_c}}
       \widetilde{\bm v}(r,c)\widetilde H_q(r,c).
 \label{eq:shape-incoming-sum}
\end{equation}
A standard $m_h$-round sumcheck for the individual-degree-two summand
$\widetilde{\bm v}\,\widetilde H_q$ reduces this claim to
\begin{equation}
 \tau=\kappa\widetilde H_q(\bm s_h),
 \qquad
 \kappa:=\widetilde{\bm v}(\bm s_h),
 \qquad \bm s_h\in\Fq^{m_h}.
\end{equation}
Here $(\bm s_h,\tau)$ is transcript-derived and $\kappa$ is computed by the
verifier.  No division is required, including when $\kappa=0$; the remainder
of $\Pi_1$ reduces this terminal claim to the successor PESAT relation.

For $\mathbb E\in\{\Fq,\K\}$, Boolean address $z\in\bitsset^m$, and
$\bm x\in\mathbb E^m$, define
\begin{equation}
 \eq_{\mathbb E}(z,\bm x):=
 \prod_{j=0}^{m-1}
 \bigl((1-z_j)(1-x_j)+z_jx_j\bigr).
 \label{eq:typed-equality-polynomial}
\end{equation}
We omit the subscript when the field is clear.  Writing
$\bm s_h=(\bm s_r,\bm s_c)$, set
$a_r:=\eq_{\Fq}(r,\bm s_r)$ and
$a'_c:=\kappa\eq_{\Fq}(c,\bm s_c)$.  Then
\begin{equation}
 W[r,c]:=\kappa\eq_{\Fq}((r,c),\bm s_h)=a_ra'_c,
 \qquad
 \tau=\sum_c a'_c\sum_r a_rH_q[r,c].
\end{equation}
Thus only the derived equality weights $W$, not the arbitrary input $V$, are
separable.  Since $\sum_r a_r=1$, the row factor remains nonzero even when
$\kappa=0$.

\smallskip
\noindent\textbf{Bounded integer-lifting and grand-product phase.}
Using the canonical lifts from Section~\ref{sec:core-parameters}, define for
every column $c$ the exact fold
\begin{equation}
 \widehat\mu_c
 :=\sum_{r<2^{d_r}}\gamma_rH_{\mathbb Z}[r,c]
 \in[0,2^{127}).
\end{equation}
The configured precondition $2^{d_r}q<2^{127}$ implies this bound because
$0\le\gamma_r<q$ and $H_{\mathbb Z}[r,c]\in\{0,1\}$.  The prover supplies
candidates $\mu_c$ and $\nu_c$.  The verifier checks
$\mu_c\in[0,2^{127})$, $\nu_c=g^{\mu_c}$, and the reconstruction
\begin{equation}
 \tau=\sum_c a'_c\pi_q(\mu_c).
\label{eq:pi1-fold-reconstruction}
\end{equation}

Since $H$ is binary, the exact fold also satisfies
\begin{equation}
 g^{\widehat\mu_c}
 =\prod_r\left(1+
    \bigl(g^{\gamma_r}-1\bigr)H_{\K}[r,c]\right).
\end{equation}
The successor PESAT relation equates this product with
$\nu_c$.  Hence $g^{\widehat\mu_c}=g^{\mu_c}$; because both exponents lie
in $[0,2^{127})$ and $\operatorname{ord}(g)=2^{128}-1$, injectivity gives
$\widehat\mu_c=\mu_c$.  Together with the input sumcheck and
global reconstruction, this establishes
\begin{equation}
 \mathcal R_{\rm LinBitsDual}[\bm h]
 \xRightarrow{\ \Pi_1\ }
 \mathcal R_{\rm PESAT}[\bm h].
 \label{eq:edge-linbits-pesat}
\end{equation}

\subsection{The F2Z PESAT relation}

Set $y_r:=g^{\gamma_r}$.  Following the paper, define the F2Z grand-product
polynomial for every column $c$ by
\begin{equation}
 Q_{\mathbb{B}\mathrm{it}\mathbb{Z},c}
   (\bm y,\bm\nu,H_{\K})
 :=\prod_r\left(1+
       (y_r-1)H_{\K}[r,c]\right)
   -\nu_c.
 \label{eq:f2z-pesat-polynomial}
\end{equation}
The underlying paper-style PESAT relation is
\begin{align}
 &\operatorname{PESAT}
   (\K,Q_{\mathbb{B}\mathrm{it}\mathbb{Z}})
 \notag\\[-2pt]
 &\quad:=
 \Bigl\{((\bm y,\bm\nu),\bot;H_{\K}):
 \bm y\in\K^{2^{d_r}},\quad
 \bm\nu\in\K^{2^{d_c}},\quad
 H_{\K}\in\{0_{\K},1_{\K}\}^{2^{d_r}\times2^{d_c}},\quad
 Q_{\mathbb{B}\mathrm{it}\mathbb{Z},c}
   (\bm y,\bm\nu,H_{\K})=0
 \quad\text{for every }c\Bigr\}.
 \label{eq:paper-pesat-relation}
\end{align}
F2Z carries the virtualized commitment invariant alongside this polynomial
relation:
\begin{align}
 \mathcal R_{\rm PESAT}[\bm h]:=
 \Bigl\{((\bm y,\bm\nu,\bm M),O;\bm f):\;&
 ((\bm y,\bm\nu),\bot;H_{\K})
   \in\operatorname{PESAT}
   (\K,Q_{\mathbb{B}\mathrm{it}\mathbb{Z}}),\notag\\[-2pt]
 &\operatorname{vec}(H)=\bm M\pi_2(\bm f),\notag\\[-2pt]
 &\text{Equations~\eqref{eq:virtual-witness} and
   \eqref{eq:overview-commitment} hold}
 \Bigr\}.
 \label{eq:rel-pesat-h}
\end{align}
This is the virtualized lifting of the paper's F2Z PESAT relation, not a
PIOP relation.  Its public input is derived inside $\Pi_1$, and
its witness remains $\bm f$; its polynomial constraints are evaluated on the
derived virtual view $H_{\K}$.
The shared binding in Equations~\eqref{eq:virtual-witness} and
\eqref{eq:overview-commitment} ties that view to $\bm f$.
Unlike a standard encoded PESAT relation, its oracle still encodes $F$, not
$H$; this is why the following reduction never queries $O$.

\paragraph{Direct reduction: layered grand-product GKR
($\Pi_{\rm GKR}$).}
Define the input layer
\begin{equation}
 L_{d_r}[r,c]
 :=1+(y_r-1)H_{\K}[r,c]
 \label{eq:pesat-input-layer}
\end{equation}
and recursively multiply adjacent row entries:
\begin{equation}
 L_i[\bm b,c]
 :=L_{i+1}[\bm b,0,c]
   L_{i+1}[\bm b,1,c],
 \quad \bm b\in\bitsset^i,\quad 0\le i<d_r.
 \label{eq:pesat-product-layers}
\end{equation}
Thus $L_0[c]=\nu_c$.  GKR verifies these multiplication
layers from the claimed outputs back to the input layer.  Its verifier
returns public weights $\bm\omega_{\xi}$ and a target $\sigma$, where
$\bm\omega_{\xi}\in\K^{n_h}$ and $\sigma\in\K$:
\begin{equation}
 (\tr',\bm\omega_{\xi},\sigma)
 \leftarrow
 \mathsf{GKR.Verify}
 (\mathsf{pp}_{\rm GKR},\mathcal I_{\rm GKR},
  \pi_{\rm GKR};\tr).
 \label{eq:gkr-verifier-contract}
\end{equation}
The resulting terminal claim is a weighted sum of entries of $H_{\K}$:
\begin{equation}
 \sum_{z\in\bitsset^{m_h}}
   \omega_{\xi,z}H_{\K}[z]=\sigma.
 \label{eq:gkr-terminal-weighted-sum}
\end{equation}
Thus GKR implements the direct reduction
\begin{equation}
 \mathcal R_{\rm PESAT}[\bm h]
 \xRightarrow{\ \Pi_{\rm GKR}\ }
 \mathcal R_{\rm sum}^{h}.
 \label{eq:edge-pesat-sum}
\end{equation}
Both values are verifier-derived from the GKR transcript; all terminal
contributions that do not multiply an entry of $H$ are moved into $\sigma$.
GKR reads $H_{\K}$ through its circuit oracle but neither queries $O$ nor
changes the commitment invariant.

\subsection{The weighted-sum relation}

The weighted-sum relation records the terminal linear claim produced by GKR:
\begin{align}
 \mathcal R_{\rm sum}^{h}:=
 \Bigl\{((\bm\omega_{\xi},\sigma,\bm M),O;\bm f):\;&
  \sum_{z\in\bitsset^{m_h}}
    \omega_{\xi,z}H_{\K}[z]=\sigma,\notag\\[-2pt]
 &\operatorname{vec}(H)=\bm M\pi_2(\bm f),\notag\\[-2pt]
 &\text{Equations~\eqref{eq:virtual-witness} and
   \eqref{eq:overview-commitment} hold}
 \Bigr\}.
 \label{eq:rel-sum-h}
\end{align}

\paragraph{Direct reduction: row LinCheck ($\Pi_{\rm MLE}$).}
The row protocol $\mathsf{LinCheck}_H$ folds the weighted claim into one
evaluation point $\bm r_h\in\K^{m_h}$ and one claimed value $\eta\in\K$.
Its immediate successor is
\begin{equation}
 \mathcal R_{\rm sum}^{h}
 \xRightarrow{\ \Pi_{\rm MLE}\ }
 \mathcal R_{\rm eval}^{h}.
 \label{eq:edge-sum-eval}
\end{equation}
The proof fragment for this reduction is denoted
$\pi_{\rm MLE}$.  The fixed GKR wire record carries
$(\pi_{\rm GKR},\pi_{\rm MLE})$ in that order; the semantic
split introduces no new transcript event.

\subsection{The MLE-evaluation relation}

The MLE-evaluation relation is
\begin{align}
 \mathcal R_{\rm eval}^{h}:=
 \Bigl\{((\bm r_h,\eta,\bm M),O;\bm f):\;&
  \widetilde H_{\K}(\bm r_h)=\eta,\notag\\[-2pt]
 &\operatorname{vec}(H)=\bm M\pi_2(\bm f),\notag\\[-2pt]
 &\text{Equations~\eqref{eq:virtual-witness} and
   \eqref{eq:overview-commitment} hold}
 \Bigr\}.
 \label{eq:rel-eval-h}
\end{align}
Using the $\K$-valued equality polynomial from
Equation~\eqref{eq:typed-equality-polynomial}, the MLE claim can be written as
\begin{equation}
 \widetilde H_{\K}(\bm r_h)
 =\sum_{z\in\bitsset^{m_h}}
   \eq(z,\bm r_h)H_{\K}[z]
 =\eta.
 \label{eq:virtual-mle-expansion}
\end{equation}

\paragraph{Direct reduction: adjoint pullback through
$\bm M_{\K}^{\mathsf T}$.}
Set
\begin{equation}
 \bm a[z]:=\eq(z,\bm r_h).
 \label{eq:virtual-adjoint-coefficient}
\end{equation}
Since $H_{\K}=\bm M_{\K}\bm f_{\K}$, the MLE claim is rewritten directly as
\begin{equation}
 \eta
 =\left\langle\bm a,H_{\K}\right\rangle_{\K}
 =\left\langle\bm a,\bm M_{\K}\bm f_{\K}\right\rangle_{\K}
 =\left\langle\bm M_{\K}^{\mathsf T}\bm a,
                 \bm f_{\K}\right\rangle_{\K}.
 \label{eq:virtual-adjoint-rewrite}
\end{equation}
Set $\bm a_f:=\bm M_{\K}^{\mathsf T}\bm a$.  This is a deterministic,
zero-message reduction whose immediate successor is
\begin{equation}
 \mathcal R_{\rm eval}^{h}
 \equiv_{\bm M_{\K}^{\mathsf T}}
 \LIN(\K,\mathcal C,\Pack).
 \label{eq:edge-eval-lin}
\end{equation}

\subsection{The commitment-facing linear relation}

The commitment-facing relation is the encoded linear relation named
$\LIN(\K,\mathcal C,\Pack)$ in the F2Z construction:
\begin{align}
 \LIN(\K,\mathcal C,\Pack):=
 \Bigl\{((\bm a_f,\eta),O;\bm f):\;&
  \bm a_f\in\K^{n_f},\quad
  \left\langle\bm a_f,\bm f_{\K}\right\rangle_{\K}=\eta,
  \notag\\[-2pt]
 &O=\Enc_{\mathcal C}(\Pack(F)),\quad
  \operatorname{vec}(F)=\pi_2(\bm f)
 \Bigr\}.
 \label{eq:rel-lin-f}
\end{align}
The adjoint produces the instance $((\bm a_f,\eta),O;\bm f)$ of this relation.
Section~\ref{sec:interaction} applies the structured ring switch directly to
this claim using the canonical decomposition from
Section~\ref{sec:ring-switch-coefficients}, then invokes the recursive Ligerito
opening.  No post-adjoint LinCheck is used, and no reduction from $\LIN$ to the
trivial relation is displayed in this section.

\paragraph{Complete direct-reduction graph.}
Collecting the immediate successors defined above gives
\begin{equation}
 \begin{aligned}
 \mathcal R_{\rm LinBitsDual}[\bm h]
 &\xRightarrow{\Pi_1}
   \mathcal R_{\rm PESAT}[\bm h]
 \xRightarrow{\Pi_{\rm GKR}}
   \mathcal R_{\rm sum}^{h}
 \\[2pt]
 &\xRightarrow{\Pi_{\rm MLE}}
   \mathcal R_{\rm eval}^{h}
 \equiv_{\bm M_{\K}^{\mathsf T}}
   \LIN(\K,\mathcal C,\Pack).
 \end{aligned}
 \label{eq:solid-reduction-graph}
\end{equation}
