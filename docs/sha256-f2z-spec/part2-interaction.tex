\section{Prover--verifier interaction}
\label{sec:sha-interaction}

\newcommand{\piq}{\pi_q}

This section instantiates the relation reductions as matched prover/verifier
moves.  A \channelN{record} is canonical-read and immediately absorbed; a
\channelH{record} is never absorbed and remains pending until its predicate
succeeds; a squeeze is a derived transcript event.  Every NARG message below
must be absorbed before the challenge that depends on it, and no squeeze is
permitted while a Hint is pending.  The two proof streams and all primitive
encodings are those of the F2Z PCS v1 profile.
The sole NARG exception is a PCS PoW nonce: parse it, permit no intervening
event, verify its predicate, and only then absorb it.
For the two SHA application squeeze tags, a field scalar is sampled by
requesting 16 bytes with auxiliary payload $\operatorname{LE}_{32}(a)$ for
attempts $a=0,1,\ldots$, masking the high 28 bits of the little-endian
integer, and accepting the first result below $q$; every attempt remains in
the transcript.  The transcript transition commits to each raw 16-byte XOF
result before the verifier masks the local candidate.  Rejection occurs before
attempt $2^{32}$; the $u_{32}$ counter never wraps.

\clgreset

\begin{step*}{P0/V0 --- Fix the input and bind the commitment}{%
  \clgnew{S}{$A_xh=0$ over $\mathbb Z$}%
  \clgnew{H}{$\pi_2(h)=h_0+T\operatorname{vec}(\pi_2(D_f))$}%
  \clgnew{C}{$\rho=\Root(\Enc(\Pack(B_f)))$}%
}
\pvpair{P0.}{Derive
  $h=\lift(h_0+T\operatorname{vec}(\pi_2(D_f)))$; encode and
  Merkle-commit only $D_f$, publishing
  $\rho$.}
 {V0.}{Validate the public SHA chain, both tensor layouts, the application,
  candidate virtual-core, GKR and code profile identifiers, and $\rho$.}
The initial obligations are
\begin{equation}
 A_xh=0\ \text{in }\mathbb Z,
 \qquad \pi_2(h)=h_0+T\operatorname{vec}(\pi_2(D_f)),
 \qquad \rho=\Root(\Enc(\Pack(B_f))).
 \label{eq:sha-initial-claims}
\end{equation}
Here $T$ is the statement-derived SHA tap matrix and $h_0$ its affine
column.  Tag \code{0x0001} binds \coreprofileid, while tag \code{0x0002}
binds \profileid, the subordinate identifiers, the statement digest, shape,
generator, and root; V0 absorbs them in that order
before any challenge.  Equation~\eqref{eq:sha-initial-claims} fixes the three
claims carried by the ledger; publishing a root alone does not discharge $C$.
\end{step*}

\begin{step}{P1/V1 --- Multilinearize the constraint rows}{%
  \clgupd{S}{$Q(\beta)=0$ on $\bits^{d_A}$}%
}
\pvpair{P1.}{Send nothing.}
 {V1.}{Pad the live rows canonically and derive their multilinear residual
  $Q$.}
The exact rewrite is
\begin{equation}
 Q(Z):=\sum_{\beta\in\bits^{d_A}}
  \eqp(\beta,Z)\langle A_x[\beta,*],h\rangle_{\mathbb Z},
 \qquad A_xh=0\Longleftrightarrow Q|_{\bits^{d_A}}=0.
 \label{eq:sha-multilinear-residual}
\end{equation}
Equation~\eqref{eq:sha-multilinear-residual} is deterministic: P1/V1 has no
message, absorb, challenge, or soundness loss.
\end{step}

\begin{step}{P2/V2 --- Project the residuals into $\Fq$}{%
  \clgupd{S}{$\bar A\piq(h)=0$}%
}
\pvpair{P2.}{Send nothing.}
 {V2.}{Set $\bar A=\piq(A_x)$ and check the statement-derived residual
  bound before continuing.}
The mandatory gate is
\begin{equation}
 q>2\max_{\beta\in\bits^{d_A}}\sum_j|A_x[\beta,j]|.
 \label{eq:sha-projection-gate}
\end{equation}
Under Equation~\eqref{eq:sha-projection-gate}, projection is injective on every
reachable bit-witness residual and loses no soundness; otherwise V2 rejects.
\end{step}

\begin{step}{P3/V3 --- Collapse the public constraint rows}{%
  \clgdone{S}%
  \clgnew{$E_0$}{$\langle v,\piq(h)\rangle=0$}%
}
\pvpair{P3.}{Send nothing.}
 {V3.}{Squeeze $r_{\rm con}\in\Fq^{d_A}$ and compute the public collapsed
  coefficient.}
For coordinate $i=0,\ldots,d_A-1$, V3 uses tag \code{0x1001}, scope
$(\bot_{32},\bot_{32},i)$, and the application field sampler above.
It then derives
\begin{equation}
 v:=\bar A^{\mathsf T}\eqp(\cdot,r_{\rm con}),
 \qquad E_0:\ \langle v,\piq(h)\rangle_{\Fq}=0.
 \label{eq:sha-row-collapse}
\end{equation}
The first squeeze occurs only after V2.  Equation~\eqref{eq:sha-row-collapse}
has error at most $d_A/q$.
\end{step}

\begin{step}{P4/V4 --- Shape the claim by sumcheck}{%
  \clgdone{$E_0$}%
  \clgnew{$E_{\rm eq}$}{$c\mle{h_q}(r_h)=\tau$}%
}
\pvpair{P4.}{For $i=0,\ldots,d_h-1$, send the degree-two polynomial $g_i$ under NARG
  tag \code{0x1002} and scope $(\bot_{32},\bot_{32},i)$.}
 {V4.}{Absorb $g_i$, check its Boolean sum, then squeeze $r_i\in\Fq$ under
  tag \code{0x1101} at the same scope.}
Put $h_q:=\piq(h)$.  The tag-\code{0x1002} payload is exactly the three
canonical values $g_i(0),g_i(1),g_i(2)$, without a vector prefix.  Starting
at $\sigma_{-1}=0$,
the round and terminal rules are
\begin{equation}
 \begin{aligned}
 g_i(0)+g_i(1)&=\sigma_{i-1},& \sigma_i&=g_i(r_i),\\
 r_h&=(r_0,\ldots,r_{d_h-1}),&
 \sigma_{-1}&=0,\quad \tau=\sigma_{d_h-1},\quad c=\mle{v}(r_h).
 \end{aligned}
 \label{eq:sha-app-sumcheck}
\end{equation}
Here $d_h=t_h+s=m_h$.  Thus
Equation~\eqref{eq:sha-app-sumcheck} leaves
$E_{\rm eq}:c\mle{h_q}(r_h)=\tau$, with error at most $2d_h/q$.
Writing $r_h=(r_{\rm row},r_{\rm col})\in\Fq^{t_h}\times\Fq^s$, V4 derives
$w_b=\can(\eqp(b,r_{\rm row}))$, $w'_c=c\eqp(c,r_{\rm col})$, and $y=\tau$.
It then absorbs derived tag \code{0x1003}, at scope
$(\bot_{32},\bot_{32},\bot_{32})$, with
payload
\begin{equation}
 \operatorname{LE}_{32}(t_h)\parallel
 \operatorname{LE}_{32}(s)\parallel
 \operatorname{Vec}_{\Fq}(r_h)\parallel
 \operatorname{Enc}_{\Fq}(c)\parallel
 \operatorname{Enc}_{\Fq}(\tau).
 \label{eq:sha-virtual-seam}
\end{equation}
The vector in Equation~\eqref{eq:sha-virtual-seam} has its canonical
little-endian $u_{32}$ count and coordinates in transcript order.  Tag
\code{0x1003} replaces the direct profile's mandatory
\code{0x0003}/$\mathsf{CoreSeamV1}$ and must precede the first
tag-\code{0x2001} record.  This replacement is legal only under
\coreprofileid; the direct or an unknown core profile rejects before this event.
The printed identifier is only a candidate namespace: until G6 and G7 freeze
both ledgers, it must not resolve to an accepting core profile.
\end{step}

\begin{step}{P5/V5 --- Expose the integer column folds}{%
  \clgdone{$E_{\rm eq}$}%
  \clgnew{T}{$u_c=\langle w,D_h[*,c]\rangle_{\mathbb Z}$}%
}
\pvpair{P5.}{Compute $u_c$ and $\nu_c=g^{u_c}$ for every virtual column;
  send $\nu$ as NARG \code{0x2001}, then $u$ as Hint \code{0x2081}.}
 {V5.}{Absorb $\nu$; range-check and authenticate the pending $u$ Hint,
  then check its reconstruction of $E_{\rm eq}$.}
For the profile's single active chunk $\ell=0$, the required checks are
\begin{equation}
 u_c=\sum_b w_bD_h[b,c],\qquad \nu_c=g^{u_c},\qquad
 \sum_c w'_c\piq(u_c)=y,\qquad 0\le u_c<2^{127}.
 \label{eq:sha-column-folds}
\end{equation}
V5 checks every verifier-visible equality in
Equation~\eqref{eq:sha-column-folds}, including $g^u=\nu$ and the full-order
condition on $g$, before clearing the Hint.  No squeeze may occur earlier.
Both vectors have exactly $2^s=128$ entries and scope
$(0,\bot_{32},\bot_{32})$.  The remaining first equality is the
claim reduced by P6--P8.
\end{step}

\vspace*{0.10in}
\begin{step}{P6/V6 --- Move the folds into products}{%
  \clgdone{T}%
  \clgnew{P}{$\prod_bL[b,c]=\nu_c$}%
}
\pvpair{P6.}{Send no additional field.}
 {V6.}{Derive the leaf family, then squeeze the initial forest point only
  after V5 has cleared its Hint.}
The leaf and initial claim are
\begin{equation}
 L[b,c]:=1+(g^{w_b}-1)D_h[b,c],\qquad
 \prod_bL[b,c]=\nu_c,\qquad
 e_0:=\mle{\nu}(\zeta).
 \label{eq:sha-product-claim}
\end{equation}
V6 obtains $\zeta\in\K^s$ with the inherited $\K$ sampler and tag
\code{0x2101}, scope
$(0,\bot_{32},i)$, in increasing $0\le i<s$.  The exponent range makes the first
reduction in Equation~\eqref{eq:sha-product-claim} exact; the random column
compression is charged to the partial-core error.
\end{step}

\begin{step}{P7/V7 --- Reduce the merged product forest}{%
  \clgdone{P}%
  \clgnew{L}{$e_d+1=\langle \Lambda,(D_h)_{z_c}\rangle_{\K}$}%
}
\pvpair{P7.}{Stream the selected GKR profile's sumcheck and child messages.}
 {V7.}{Check and absorb every message before its challenge, returning the
  final leaf-boundary point and value.}
The one core-fixed GKR event is NARG tag \code{0x3001}: the phase-$c$ claimed
sum is absorbed after the phase-link check and before the first phase-$c$
round.  The selected GKR profile owns NARG tags \code{0x3002--0x30ff} and
squeeze tags \code{0x3100--0x31ff}; tag \code{0x3081} is forbidden.  At each
layer terminal, V7 checks the child pair before absorbing it and squeezing its
glue challenge.  The returned claim is
\begin{equation}
 e_d+1=\sum_{b,c}\eqp(b,z_r)\eqp(c,z_c)
       (g^{w_b}-1)D_h[b,c]=:\langle \Lambda,(D_h)_{z_c}\rangle_{\K}.
 \label{eq:sha-gkr-boundary}
\end{equation}
Equation~\eqref{eq:sha-gkr-boundary} is accepted only if the statement-bound
GKR identifier resolves to a byte-exact ledger; an unknown ledger rejects.
\end{step}

\begin{step}{P8/V8 --- Reduce the boundary to one virtual evaluation}{%
  \clgdone{L}%
  \clgnew{$E_H$}{$\mle{D_h}(r_*,z_c)=\eta$}%
}
\pvpair{P8.}{Run the degree-two row-only pre-sumcheck.}
 {V8.}{Check each Boolean sum before its row challenge; require the terminal
  public factor to be nonzero before dividing.}
Let $\widetilde\Lambda$ be the multilinear extension of the public row
coefficient returned by P7/V7, and let $\theta$ be the terminal value of the
row-only sumcheck.  The intended output is
\begin{equation}
 \widetilde\Lambda(r_{\rm row}^{*})\ne0,\qquad
 \eta:=\frac{\theta}{\widetilde\Lambda(r_{\rm row}^{*})},\qquad
 E_H:\ \mle{D_h}(r_{\rm row}^{*},z_c)=\eta.
 \label{eq:sha-virtual-evaluation}
\end{equation}
Equation~\eqref{eq:sha-virtual-evaluation} has no remaining column variable.
\textbf{\hyperlink{gap:sha-g6}{G6 (partial-core event ledger).}}
The PCS core assigns no standalone tags, payload codec, scopes, or cardinality
to this pre-sumcheck.  Until the selected virtual-GKR identifier publishes
those events inside its reserved ledger and states their soundness error, V8
must reject before reading a P8 message; it must not infer structure order.
That ledger must include the nonzero check in
Equation~\eqref{eq:sha-virtual-evaluation}; a zero terminal factor rejects.
This is a profile-closure condition, not an optional implementation detail.
\end{step}

\begin{step}{P9/V9 --- Apply the full virtual adjoint}{%
  \clgdone{$E_H$}%
  \clgdone{H}%
  \clgnew{$E_f$}{$\langle u,f_{\K}\rangle=\eta+\kappa_0$}%
}
\pvpair{P9.}{Send nothing.}
 {V9.}{Compute the statement-bound full transpose $T_{\K}^{*}$ and its affine
  correction; do not squeeze or absorb.}
Write $f_{\K},h_{\K},h_{0,\K}$ for the coefficientwise embeddings of
$\operatorname{vec}(D_f),h,h_0$ into $\K$, and write $T_{\K}$ for the
coefficientwise extension of $T$.  With
$q_H=\eqp(\cdot,(r_{\rm row}^{*},z_c))$, V9 derives
\begin{equation}
 \begin{aligned}
 \eta&=\langle q_H,h_{\K}\rangle_{\K}
     =\langle T_{\K}^{*}q_H,f_{\K}\rangle_{\K}
       +\langle q_H,h_{0,\K}\rangle_{\K},\\
 u&:=T_{\K}^{*}q_H,\qquad
 \kappa_0:=-\langle q_H,h_{0,\K}\rangle_{\K},\\
 E_f&:\quad \langle u,f_{\K}\rangle_{\K}=\eta+\kappa_0.
 \end{aligned}
 \label{eq:sha-full-adjoint}
\end{equation}
Equation~\eqref{eq:sha-full-adjoint} is exact and zero-message.  A tap-closure
optimization is admissible only after proving equality to the full transpose;
a columnwise $T_0$ shortcut rejects.
\end{step}

\begin{step}{P10/V10 --- Adapt and ring-switch the committed claim}{%
  \clgdone{$E_f$}%
  \clgnew{$E_P$}{$\langle B,P\rangle_{\K}=\beta$}%
}
\pvpair{P10.}{Supply the registered adapter's partial evaluations.}
 {V10.}{Verify them against $E_f$, then derive one packed coefficient and
  target against $P=\Pack(B_f)$.}
Write each source point as $r_f^{(a)}=(r_b^{(a)},r_c^{(a)})$ in semantic
$(b,c)$ order, and set
$\widehat r_f^{(a)}:=(r_c^{(a)},r_b^{(a)})$,
$F:=\Pi_f f_{\K}$, and $\widehat u:=\Pi_f u$.  Thus $F$ is in the inherited
PCS $(c,b)$ order and $\langle\widehat u,F\rangle=\langle u,f_{\K}\rangle$.
The required adapter contract is
\begin{equation}
 \widehat u=\sum_a\lambda_a\eqp(\cdot,\widehat r_f^{(a)}),\qquad
 E_f:\ \sum_a\lambda_a\mle{F}(\widehat r_f^{(a)})=\eta+\kappa_0
 \quad\Longrightarrow\quad
 E_P:\ \langle B,P\rangle_{\K}=\beta.
 \label{eq:sha-ring-adapter-contract}
\end{equation}
The direct PCS events \code{0x4001}, \code{0x4101}, and \code{0x4102} cover
one MLE-shaped source per active chunk, not the multi-evaluation input in
Equation~\eqref{eq:sha-ring-adapter-contract}.
\textbf{\hyperlink{gap:sha-g7}{G7 (ring-adapter ledger).}}
No registered profile currently fixes the extra NARG fields, term scopes,
checks, squeeze deadlines, or composed error for this adapter.  V10 must reject
before consuming an adapter field unless its identifier resolves to that
ledger.  Tags \code{0x4001/0x4101/0x4102} must not be silently reinterpreted.
Only closure of G7 may produce the single deterministic claim $(B,\beta)$
consumed by P11.
\end{step}

\begin{step}{P11/V11 --- Open the packed claim}{%
  \clgdone{$E_P$}%
  \clgdone{C}%
}
\pvpair{P11.}{Send the recursive proof and openings.}
 {V11.}{Verify them against $\rho$ and exhaust both streams.}
For this committed shape, the packed opening target is exactly
\begin{equation}
 m_p=t_f+s-7=t_f,\qquad
 B:\{0,1\}^{t_f}\longrightarrow\K,\qquad
 \mathsf{payload}_{5001}=\operatorname{LE}_{32}(t_f)
   \parallel\operatorname{Enc}_{\K}(\beta).
 \label{eq:sha-opening-target}
\end{equation}
V11 first checks the domain in Equation~\eqref{eq:sha-opening-target}, then
absorbs derived tag \code{0x5001} before the recursive helper.  The next
tag-\code{0x5002} root must equal $\rho$.  V11 then
reuses the PCS ledger without alteration: roots \code{0x5002/0x5003},
sumcheck messages \code{0x5004}, OOD values \code{0x5005}, final vector
\code{0x5006}, fold/query nonces \code{0x5007/0x5008}, Merkle Hints
\code{0x5080--0x5082}, and squeezes \code{0x5101--0x5106}.  Every Hint is
authenticated before a later event, and both streams, every nested slice, and
the host container must end exactly.

With the two unresolved adapters named rather than assumed, the composed
soundness statement is
\begin{equation}
 \varepsilon_{\rm SHA\text{-}F2Z}
 \le \frac{d_A+2d_h}{q}
   +\varepsilon_{\rm partial}
   +\varepsilon_{\rm vring}
   +\varepsilon_{\rm batch}
   +\varepsilon_{\rm open}.
 \label{eq:sha-composed-error}
\end{equation}
Here $\varepsilon_{\rm partial}$ includes P6 column compression, P7 GKR, and
P8 pre-sumcheck; $\varepsilon_{\rm vring}$ belongs to P10.
Equation~\eqref{eq:sha-composed-error} is symbolic until
\hyperlink{gap:sha-g6}{G6} and \hyperlink{gap:sha-g7}{G7} publish their proofs
and numeric bounds.  P1, P2, P5, and P9 are exact under their stated reject
gates.
\end{step}
