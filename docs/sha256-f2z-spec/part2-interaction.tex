\section{Prover--verifier interaction}
\label{sec:sha-interaction}

\newcommand{\piq}{\pi_q}

Read each table from top to bottom.  A blank Prover or Verifier cell means that
actor is idle.  The middle column records the event order.  An $\mathsf N$ or
$\mathsf H$ arrow carries a NARG record or Hint; $\mathsf D$ marks a derived
challenge or public value and carries no proof bytes.

Every NARG record is absorbed before its dependent challenge.  A Hint is never
absorbed and must be checked before the next squeeze.  The sole exception to
immediate NARG absorption is a PCS PoW nonce: parse it, permit no intervening
event, verify its predicate, and then absorb it.  Both proof streams and all
primitive encodings are inherited from the F2Z PCS v1 profile.
Tags \code{0x1001} and \code{0x1101} sample a field scalar by requesting
16 bytes with auxiliary payload $\operatorname{LE}_{32}(a)$ for attempts
$a=0,1,\ldots$.  Each raw XOF result enters the transcript before its high
28 bits are masked.  The first value below $q$ is accepted; rejection occurs
before attempt $2^{32}$, so the $u_{32}$ counter never wraps.

\begin{pvstepzero}{Fix the input and bind the commitment}{%
  \clgnew{S}{$A_xh=0$ over $\mathbb Z$}%
  \clgnew{H}{$\pi_2(h)=h_0+T\operatorname{vec}(\pi_2(D_f))$}%
  \clgnew{C}{$\rho=\Root(O)$}%
}
\begin{pvexchange}
\Pturn{Derive
  \(h=\lift(h_0+T\operatorname{vec}(\pi_2(D_f)))\), then set
  $O:=\Enc(\Pack(B_f))$ and Merkle-commit only $D_f$.}
\Psend{\Dbadge{\rho}}
\Vturn{Validate the public SHA chain, both tensor layouts, all application,
  virtual-core, GKR, and code identifiers, and $\rho$.  Require
  $\operatorname{ord}(g)=2^{128}-1$.}
\Vturn{Absorb tag \code{0x0001}, followed by tag \code{0x0002}, before the
  first challenge.}
\end{pvexchange}
Here $T$ is the statement-derived SHA tap matrix and $h_0$ its affine column.
Tag \code{0x0001} binds the virtual-core identifier \coreprofileid.
Tag \code{0x0002} binds the application profile, subordinate identifiers,
statement digest, shape, generator, and root.
Publishing $\rho$ does not discharge $C$.
\end{pvstepzero}

\begin{pvstep}{Multilinearize the constraint rows}{%
  \clgupd{S}{$Q(\beta)=0$ on $\bits^{d_A}$}%
}
\begin{pvexchange}
\Pturn{On witness $h$, form
\(
 Q(Z):=\sum_{\beta\in\bits^{d_A}}
  \eqp(\beta,Z)\langle A_x[\beta,*],h\rangle_{\mathbb Z}.
\)}
\Vturn{Validate the canonical row padding and bind the formal rewrite
\(
 A_xh=0\iff Q|_{\bits^{d_A}}=0.
\)
No witness evaluation is performed.}
\end{pvexchange}
P1/V1 has no message, absorb, challenge, or soundness loss.
\end{pvstep}

\begin{pvstep}{Project the residuals into $\Fq$}{%
  \clgupd{S}{$\bar A\piq(h)=0$}%
}
\begin{pvexchange}
\Vturn{Set $\bar A:=\piq(A_x)$ and require
\(
 q>2\max_{\beta\in\bits^{d_A}}\sum_j|A_x[\beta,j]|.
\)}
\end{pvexchange}
This bound makes projection injective on every reachable residual; otherwise
V2 rejects.
\end{pvstep}

\begin{pvstep}{Collapse the public constraint rows}{%
  \clgdone{S}%
  \clgnew{$E_0$}{$\langle v,\piq(h)\rangle=0$}%
}
\begin{pvexchange}
\Vturn{Squeeze $r_{\rm con}\in\Fq^{d_A}$ with tag \code{0x1001} in increasing
  coordinate order.}
\Vsend{\Dbadge{r_{\rm con}}}
\Pturn{Derive
\(
 v:=\bar A^{\mathsf T}\eqp(\cdot,r_{\rm con}).
\)}
\Vturn{Independently derive the same $v$ and set the target to zero.}
\end{pvexchange}
Each coordinate uses scope $(\bot_{32},\bot_{32},i)$; the reduction error is
at most $d_A/q$.
\end{pvstep}

\begin{pvstep}[2.8in]{Shape the claim by sumcheck}{%
  \clgdone{$E_0$}%
  \clgnew{$E_{\rm eq}$}{$c\mle{h_q}(r_h)=\tau$}%
}
\begin{pvexchange}
\Pturn{In round $i$, after $r_0,\ldots,r_{i-1}$, compute
  $g_i\in\Fq[X]$ with $\deg g_i\le2$.}
\Psend{\Nbadge{g_i}}
\Vturn{Read and absorb tag \code{0x1002} at
$(\bot_{32},\bot_{32},i)$, then require
\(
  g_i(0)+g_i(1)=\sigma_{i-1}.
\)}
\Vturn{Squeeze $r_i\in\Fq$ under tag \code{0x1101} at the same scope.}
\Vsend{\Dbadge{r_i}}
\Pturn{Set $\sigma_i:=g_i(r_i)$.}
\Vturn{Set $\sigma_i:=g_i(r_i)$.}
\Pturn{After the last round, derive
\(
 r_h=(r_0,\ldots,r_{d_h-1}),\quad
 \tau:=\sigma_{d_h-1},\quad c:=\mle v(r_h).
\)
and absorb the virtual seam under tag \code{0x1003}.}
\Vturn{Derive the same $(r_h,\tau,c)$, then absorb the virtual seam under tag
  \code{0x1003}.}
\end{pvexchange}
Here $h_q:=\piq(h)$, $\sigma_{-1}:=0$, and $d_h=t_h+s=m_h$; the reduction
error is at most $2d_h/q$.
Writing $r_h=(r_{\rm row},r_{\rm col})\in\Fq^{t_h}\times\Fq^s$, V4 derives
$w_b\in\{0,\ldots,q-1\}$ uniquely from
$\piq(w_b)=\eqp(b,r_{\rm row})$, and sets
$w'_c=c\eqp(c,r_{\rm col})$ and $y=\tau$.
The derived tag \code{0x1003}, at scope
$(\bot_{32},\bot_{32},\bot_{32})$, with
payload
\begin{equation}
 \operatorname{LE}_{32}(t_h)\parallel
 \operatorname{LE}_{32}(s)\parallel
 \operatorname{Vec}_{\Fq}(r_h)\parallel
 \operatorname{Enc}_{\Fq}(c)\parallel
 \operatorname{Enc}_{\Fq}(\tau).
 \label{eq:sha-virtual-seam}
\end{equation}
The vector uses its canonical $u_{32}$ count and transcript order.  For this
virtual-core profile, tag \code{0x1003} replaces $\mathsf{CoreSeamV1}$ and precedes
the first tag-\code{0x2001} record; every other profile rejects.  G6 and G7
must close the two ledgers before this profile can accept.
\end{pvstep}

\begin{pvstep}[2.4in]{Expose the integer column folds}{%
  \clgdone{$E_{\rm eq}$}%
  \clgnew{T}{$u_c=\langle w,D_h[*,c]\rangle_{\mathbb Z}$}%
}
\begin{pvexchange}
\Pturn{For each virtual column $c$, compute
\(
 u_c:=\sum_b w_bD_h[b,c],\qquad \nu_c:=g^{u_c}.
\)}
\Psend{\Nbadge{\nu}}
\Vturn{Read and absorb $\nu$ under tag \code{0x2001}.}
\Psend{\Hbadge{u}}
\Vturn{Read $u$ under tag \code{0x2081}; before any squeeze, require
$0\le u_c<2^{127}$ and $g^{u_c}=\nu_c$ for every $c$, together with
$\sum_c w'_c\piq(u_c)=y$.  Then clear the Hint.}
\end{pvexchange}
Both vectors have $2^s=128$ entries and scope
$(0,\bot_{32},\bot_{32})$; V0 already checked the order of $g$.
\end{pvstep}

\begin{pvstep}[2.15in]{Move the folds into products}{%
  \clgdone{T}%
  \clgnew{P}{$\prod_bL[b,c]=\nu_c$}%

\begin{pvexchange}
\Pturn{Form the witness leaf values
\(
 L[b,c]:=1+(g^{w_b}-1)D_h[b,c].
\)}
\Vturn{After V5 clears its Hint, instantiate the corresponding product
  relation.}
\Vturn{Squeeze $\zeta\in\K^s$ under tag \code{0x2101} in increasing
  coordinate order.}
\Vsend{\Dbadge{\zeta}}
\Pturn{Derive $e_0:=\mle\nu(\zeta)$ and initialize the product forest.}
\Vturn{Independently derive $e_0:=\mle\nu(\zeta)$.}
\end{pvexchange}
V6 obtains $\zeta\in\K^s$ with the inherited $\K$ sampler and tag
\code{0x2101}, scope
$(0,\bot_{32},i)$, in increasing $0\le i<s$.  The exponent range makes the
product reduction exact; random column compression is charged to the
partial-core error.
\end{pvstep}

\begin{pvstep}[2.3in]{Reduce the merged product forest to LIN}{%
  \clgdone{P}%
  \clgnew{$E_H$}{$\langle R,h_{\K}\rangle_{\K}=e$}%
}
\begin{pvexchange}
\Pturn{Run the registered virtual-GKR prover on the product forest.}
\Psend{\Nbadge{\pi_{\rm GKR}}}
\Vturn{Replay the registered virtual-GKR ledger, checking and absorbing every
  message before its dependent challenge.  Require complete consumption and
  return $(R,e)$.}
\end{pvexchange}
The registered profile owns NARG tags \code{0x3001--0x30ff} and squeeze tags
\code{0x3100--0x31ff}; tag \code{0x3081} is forbidden.  It returns
\begin{equation}
 E_H:\quad \langle R,h_{\K}\rangle_{\K}=e,
 \qquad h_{\K}=h_{0,\K}+T_{\K}f_{\K},
 \label{eq:sha-gkr-lin}
\end{equation}
where $R$ is verifier-derived rather than transmitted densely.
\textbf{\hyperlink{gap:sha-g6}{G6}} requires the exact P5--P7 ledger,
reduction proof, and error; an unknown ledger rejects.
\end{pvstep}

\begin{pvstep}[2.25in]{Apply the full virtual adjoint}{%
  \clgdone{$E_H$}%
  \clgdone{H}%
  \clgnew{$E_f$}{$\langle a,F\rangle_{\K}=\mu_f$}%
}
\begin{pvexchange}
\Pturn{Derive
\(
 a:=\Pi_fT_{\K}^{*}R,\qquad
 \mu_f:=e-\langle R,h_{0,\K}\rangle_{\K}.
\)}
\Vturn{Independently derive the same $(a,\mu_f)$ and the public evaluator
\(
 \mle a(r)=
 \left\langle R,
 T_{\K}\!\left(\Pi_f^{-1}\eqp(\cdot,r)\right)
 \right\rangle_{\K}.
\)}
\end{pvexchange}
For the committed bits $F:=\Pi_f f_{\K}$ in PCS $(c,b)$ order,
\begin{equation}
 a:=\Pi_fT_{\K}^{*}R,\qquad
 \mu_f:=e-\langle R,h_{0,\K}\rangle_{\K},\qquad
 E_f:\ \langle a,F\rangle_{\K}=\mu_f.
 \label{eq:sha-full-adjoint}
\end{equation}
The table gives its exact public evaluator.  A sparse evaluator is conforming
only if it is identical to the full adjoint; a columnwise shortcut rejects.
\end{pvstep}

  \clgdone{$E_f$}%
  \clgnew{LC}{$\sigma=\widetilde a(r)\mle{F}(r)$}%

\begin{pvexchange}
\Pturn{In round $i$, after $r_0,\ldots,r_{i-1}$, compute

\Psend{\Nbadge{g_i}}

$(\bot_{32},\bot_{32},i)$, then require

 g_i(0)+g_i(1)=\sigma_{i-1}.

\Vturn{Squeeze $r_i\in\K$ under tag \code{0x4103} at the same scope.}
\Vsend{\Dbadge{r_i}}
\Pturn{Set $\sigma_i:=g_i(r_i)$.}
\Vturn{Set $\sigma_i:=g_i(r_i)$.}
\end{pvexchange}
Start with $\sigma_{-1}:=\mu_f$.  At the final point

 \sigma_{m_f-1}=\widetilde a(r)\mle{F}(r).
 \label{eq:sha-lincheck-terminal}

P10/V10 checks Equation~\eqref{eq:sha-lincheck-terminal} with the first ring

\end{pvstep}

\begin{pvstep}[3.0in]{Discharge LinCheck and ring-switch}{%
  \clgdone{LC}%
  \clgnew{$E_P$}{$\langle B,P\rangle_{\K}=\beta$}%
}
\begin{pvexchange}
\Pturn{For $0\le v<128$, compute
\(
 s_v:=\mle{(B_f)_{v,\K}}(r_{\rm hi}).
\)}
\Psend{\Nbadge{s}}
\Vturn{Read and absorb $s$ under tag \code{0x4001} at
$(0,\bot_{32},\bot_{32})$.  Derive
\(
 z:=\sum_{v=0}^{127}\eqp(v,r_{\rm lo})s_v
   =\mle F(r)
\)
and require $\sigma_{m_f-1}=\widetilde a(r)z$.}
\Vturn{Only after that check, squeeze $r''\in\K^7$ under tag \code{0x4101}.}
\Vsend{\Dbadge{r''}}
\Pturn{Derive the ring-switch successor $(B_0,\beta_0)$.}
\Vturn{Independently derive the same $(B_0,\beta_0)$.}
\Vturn{Squeeze $\eta_0\in\K$ under tag \code{0x4102}.}
\Vsend{\Dbadge{\eta_0}}
\Pturn{Set $B:=\eta_0B_0$ and $\beta:=\eta_0\beta_0$.}
\Vturn{Independently derive the same $(B,\beta)$.}
\end{pvexchange}
\begin{equation}
 B:=\eta_0B_0,\qquad \beta:=\eta_0\beta_0,
 \qquad E_P:\ \langle B,P\rangle_{\K}=\beta,\qquad P=\Pack(B_f).
 \label{eq:sha-ring-successor}
\end{equation}
No extra adapter message, target, or second LinCheck exists.  The event rules
are now fixed by the PCS profile; \textbf{\hyperlink{gap:sha-g7}{G7}} is the

\end{pvstep}

\begin{pvstep}[2.4in]{Open the packed claim}{%
  \clgdone{$E_P$}%
  \clgdone{C}%
}
\begin{pvexchange}
\Pturn{Require
\(
 m_p=t_f,\quad B:\bits^{t_f}\to\K,
\)
derive
\(
 \mathsf{payload}_{5001}:=
 \operatorname{LE}_{32}(t_f)\parallel\operatorname{Enc}_{\K}(\beta),
\)
and absorb tag \code{0x5001}.}
\Vturn{Independently require the same domain and absorb the same derived tag.}
\Pturn{Run the inherited recursive opening prover for $(\rho,B,\beta)$.}
\Psend{\Splitbadge{\pi_{\rm open}}}
\Vturn{Replay the inherited opening ledger against $(\rho,B,\beta)$.}
\Vturn{Accept only if the opening succeeds, every Hint is authenticated, and
  both streams, all nested slices, and the host container end exactly.}
\end{pvexchange}
The composed soundness error is
\begin{equation}
 \varepsilon_{\rm SHA\text{-}F2Z}
 \le \frac{d_A+2d_h}{q}
   +\varepsilon_{\rm partial}
   +\frac{2m_f}{2^{128}}
   +\varepsilon_{\rm ring}
   +\varepsilon_{\rm batch}
   +\varepsilon_{\rm open}.
 \label{eq:sha-composed-error}
\end{equation}
Here $\varepsilon_{\rm partial}$ includes P6 column compression and P7 GKR;
$2m_f/2^{128}$ is the single P9 LinCheck term.
Equation~\eqref{eq:sha-composed-error} is symbolic until
\hyperlink{gap:sha-g6}{G6} and \hyperlink{gap:sha-g7}{G7} publish their proofs
and fixtures.  P1, P2, P5, and P8 are exact under their stated reject gates.
\end{pvstep}
